Whenever someone asks me what my nerdy passion project is, I tell them Minecraft Hogwarts. I have been building Hogwarts in Minecraft since 2016 in what has become my magnum opus.

## About the project

Since March 2019, I have been recreating Hogwarts Castle from *Harry Potter* in Minecraft. The project combines research into the castle’s architectural design with detailed reconstruction and interior accuracy.

Command blocks, data packs, and custom textures and models help bring the castle to life.

Alongside Hogwarts, I have built the Ministry of Magic, Azkaban, Malfoy Manor, and the Burrow.

I documented this journey on YouTube, where my channel has grown to over 38,000 subscribers and 7.5 million views. I have also produced a 34-episode tutorial series guiding viewers through the castle’s construction.

(function () {
  /* ── Scroll-reveal ── */
  var reveals = document.querySelectorAll('.pd-reveal');

  if (!('IntersectionObserver' in window)) {
    reveals.forEach(function (el) { el.classList.add('is-visible'); });
  } else {
    var obs = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (!entry.isIntersecting) return;
        entry.target.classList.add('is-visible');
        obs.unobserve(entry.target);
      });
    }, { threshold: 0.35, rootMargin: '0px 0px -8% 0px' });
    reveals.forEach(function (el) { obs.observe(el); });
  }
})();

    lastClone.id = "last-clone";
    track.appendChild(firstClone);
    track.insertBefore(lastClone, slides[0]);

    const allSlides = Array.from(track.children);
    let slideWidth = allSlides[0].getBoundingClientRect().width;

    function jumpTo(index) {
      track.style.transition = "none";
      track.style.transform = `translateX(-${slideWidth * index}px)`;
      currentIndex = index;
    }

    function moveToIndex(index) {
      track.style.transition = "transform 0.6s ease";
      track.style.transform = `translateX(-${slideWidth * index}px)`;
      currentIndex = index;
    }

    // Initial position (account for leading clone)
    jumpTo(currentIndex);

    // Handle loop wrap on transition end
    function handleTransitionEnd() {
      const currentSlide = allSlides[currentIndex];
      if (currentSlide.id === "first-clone") {
        jumpTo(1);
      } else if (currentSlide.id === "last-clone") {
        jumpTo(allSlides.length - 2);
      }
    }

    // Buttons
    nextButton?.addEventListener("click", () => {
      if (currentIndex >= allSlides.length - 1) return;
      moveToIndex(currentIndex + 1);
    });

    prevButton?.addEventListener("click", () => {
      if (currentIndex <= 0) return;
      moveToIndex(currentIndex - 1);
    });

    track.addEventListener("transitionend", handleTransitionEnd);

    // Resize: recalc width and snap without animation
    window.addEventListener("resize", () => {
      slideWidth = allSlides[0].getBoundingClientRect().width;
      jumpTo(currentIndex);
    });

    // -------------------------
    // Swipe / Drag functionality
    // -------------------------
    let isDragging = false;
    let startX = 0;
    let lastX = 0;
    let draggingDiff = 0;

    const thresholdPx = () => Math.max(40, Math.min(160, slideWidth * 0.2)); // 20% width, clamped

    function onDragStart(clientX) {
      isDragging = true;
      startX = clientX;
      lastX = clientX;
      draggingDiff = 0;
      track.style.transition = "none";
    }

    function onDragMove(clientX) {
      if (!isDragging) return;
      lastX = clientX;
      draggingDiff = lastX - startX; // + means dragging right (toward prev)
      const base = -slideWidth * currentIndex;
      track.style.transform = `translateX(${base + draggingDiff * 1}px)`;
    }

    function onDragEnd() {
      if (!isDragging) return;
      const diff = lastX - startX;
      const abs = Math.abs(diff);
      const threshold = thresholdPx();

      // Restore transition for snap/advance
      track.style.transition = "transform 0.6s ease";

      if (abs >= threshold) {
        if (diff > 0 && currentIndex > 0) {
          // dragged right -> go to previous
          moveToIndex(currentIndex - 1);
        } else if (diff < 0 && currentIndex < allSlides.length - 1) {
          // dragged left -> go to next
          moveToIndex(currentIndex + 1);
        } else {
          moveToIndex(currentIndex);
        }
      } else {
        // Not enough movement: snap back
        moveToIndex(currentIndex);
      }

      isDragging = false;
      draggingDiff = 0;
    }

    // Prefer Pointer Events (covers mouse + touch in modern browsers)
    const usePointer = "PointerEvent" in window;

    if (usePointer) {
      track.addEventListener("pointerdown", (e) => {
        // Only react to primary button / touch
        if (e.button !== undefined && e.button !== 0) return;
        track.setPointerCapture?.(e.pointerId);
        onDragStart(e.clientX);
      });

      track.addEventListener("pointermove", (e) => onDragMove(e.clientX));
      track.addEventListener("pointerup", onDragEnd);
      track.addEventListener("pointercancel", onDragEnd);
      track.addEventListener("pointerleave", () => {
        if (isDragging) onDragEnd();
      });
    } else {
      // Touch fallback (older Safari)
      track.addEventListener(
        "touchstart",
        (e) => onDragStart(e.touches[0].clientX),
        { passive: true }
      );
      track.addEventListener(
        "touchmove",
        (e) => onDragMove(e.touches[0].clientX),
        { passive: true }
      );
      track.addEventListener(
        "touchend",
        (e) => onDragEnd(),
        { passive: true }
      );

      // Optional mouse fallback
      track.addEventListener("mousedown", (e) => {
        if (e.button !== 0) return;
        onDragStart(e.clientX);
      });
      window.addEventListener("mousemove", (e) => onDragMove(e.clientX));
      window.addEventListener("mouseup", onDragEnd);
    }
  });
})();
</script>
