      <h2>The project I always come back to</h2>
      <p>I never really set out to spend years building Hogwarts. It started more simply: I loved the castle, I loved Minecraft. What began as a fun build quickly turned into something much deeper. The more I worked on it, the more I found myself researching tiny architectural details and rebuilding sections I thought I’d finished after new reference material became available.</p>
      <br>
      <p>I documented this journey on YouTube, where my channel has grown to over 40,000 subscribers and 8 million views. I have also produced a 34-episode tutorial series guiding viewers through the castle’s construction.</p>

  function applyTransform() {
    mapCanvas.style.transform =
      'translate(' + xfm.tx.toFixed(2) + 'px,' + xfm.ty.toFixed(2) + 'px) scale(' + xfm.s.toFixed(4) + ')';
    updateMarkerPositions();
  }

  function updateMarkerPositions() {
    var W = mapWrap.offsetWidth;
    var H = mapWrap.offsetHeight;
    mapOverlay.querySelectorAll('.hw-marker').forEach(function (m) {
      m.style.left = (xfm.tx + parseFloat(m.dataset.lx) / 100 * W * xfm.s) + 'px';
      m.style.top  = (xfm.ty + parseFloat(m.dataset.ly) / 100 * H * xfm.s) + 'px';
    });
  }

  /* Reposition markers after each new floor image loads (height may change) */
  mapImg.addEventListener('load', updateMarkerPositions);

  function clampTransform() {
    var ww = mapWrap.offsetWidth;
    var wh = mapWrap.offsetHeight;
    xfm.tx = clamp(xfm.tx, ww * (1 - xfm.s), 0);
    xfm.ty = clamp(xfm.ty, wh * (1 - xfm.s), 0);
  }

  function zoomAt(cx, cy, newS) {
    newS = clamp(newS, MIN_S, MAX_S);
    var r = newS / xfm.s;
    xfm.tx = cx - (cx - xfm.tx) * r;
    xfm.ty = cy - (cy - xfm.ty) * r;
    xfm.s  = newS;
    clampTransform();
    applyTransform();
  }

  /* ── Wheel zoom ─────────────────────────────────────── */
  mapWrap.addEventListener('wheel', function (e) {
    e.preventDefault();
    var rect  = mapWrap.getBoundingClientRect();
    var delta = e.deltaY > 0 ? 0.85 : 1 / 0.85;
    zoomAt(e.clientX - rect.left, e.clientY - rect.top, xfm.s * delta);
  }, { passive: false });

  /* ── Double-click resets zoom ───────────────────────── */
  mapWrap.addEventListener('dblclick', function () {
    xfm = { s: 1, tx: 0, ty: 0 };
    applyTransform();
  });

  /* ── Mouse drag ─────────────────────────────────────── */
  var drag = null;
  var dragMoved = false;

  mapWrap.addEventListener('mousedown', function (e) {
    if (e.button !== 0) return;
    drag = { x: e.clientX, y: e.clientY, tx: xfm.tx, ty: xfm.ty };
    dragMoved = false;
    mapWrap.classList.add('is-dragging');
    e.preventDefault();
  });

  window.addEventListener('mousemove', function (e) {
    if (!drag) return;
    var dx = e.clientX - drag.x;
    var dy = e.clientY - drag.y;
    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragMoved = true;
    xfm.tx = drag.tx + dx;
    xfm.ty = drag.ty + dy;
    clampTransform();
    applyTransform();
  });

  window.addEventListener('mouseup', function () {
    drag = null;
    mapWrap.classList.remove('is-dragging');
  });

  /* ── Touch: single-finger drag + two-finger pinch ───── */
  var touch = null;
  var touchMoved = false;
  var touchMarker = null;

  function touchDist(a, b) {
    var dx = a.clientX - b.clientX, dy = a.clientY - b.clientY;
    return Math.sqrt(dx * dx + dy * dy);
  }

  mapWrap.addEventListener('touchstart', function (e) {
    e.preventDefault();
    mapWrap.classList.add('is-dragging');
    touchMoved = false;
    touchMarker = e.target.closest ? e.target.closest('.hw-marker') : null;
    var rect = mapWrap.getBoundingClientRect();
    if (e.touches.length === 1) {
      touch = { fingers: 1, x: e.touches[0].clientX, y: e.touches[0].clientY,
                tx: xfm.tx, ty: xfm.ty };
    } else if (e.touches.length >= 2) {
      var mx = (e.touches[0].clientX + e.touches[1].clientX) / 2;
      var my = (e.touches[0].clientY + e.touches[1].clientY) / 2;
      touch = { fingers: 2, d: touchDist(e.touches[0], e.touches[1]), s: xfm.s,
                cx: mx - rect.left, cy: my - rect.top };
    }
  }, { passive: false });

  mapWrap.addEventListener('touchmove', function (e) {
    e.preventDefault();
    if (!touch) return;
    if (touch.fingers === 1 && e.touches.length >= 1) {
      var dx = e.touches[0].clientX - touch.x;
      var dy = e.touches[0].clientY - touch.y;
      if (Math.abs(dx) > 6 || Math.abs(dy) > 6) touchMoved = true;
      xfm.tx = touch.tx + dx;
      xfm.ty = touch.ty + dy;
      clampTransform();
      applyTransform();
    } else if (touch.fingers === 2 && e.touches.length >= 2) {
      touchMoved = true;
      var newD = touchDist(e.touches[0], e.touches[1]);
      var newS = clamp(touch.s * (newD / touch.d), MIN_S, MAX_S);
      var r    = newS / xfm.s;
      xfm.tx = touch.cx - (touch.cx - xfm.tx) * r;
      xfm.ty = touch.cy - (touch.cy - xfm.ty) * r;
      xfm.s  = newS;
      clampTransform();
      applyTransform();
    }
  }, { passive: false });

  mapWrap.addEventListener('touchend', function () {
    if (touchMarker && !touchMoved) {
      touchMarker.click();
    }
    touch = null;
    touchMoved = false;
    touchMarker = null;
    mapWrap.classList.remove('is-dragging');
  });

  /* ── Build floor tabs ────────────────────────────────── */
  HW_FLOORS.forEach(function (floor) {
    var btn = document.createElement('button');
    btn.className = 'hw-map-tab';
    btn.setAttribute('type', 'button');
    btn.textContent = floor.label;

    btn.addEventListener('click', function () {
      if (activeFloor === floor.id) return;
      tabsEl.querySelectorAll('.hw-map-tab').forEach(function (t) {
        t.classList.remove('is-active');
      });
      btn.classList.add('is-active');
      showFloor(floor);
    });

    tabsEl.appendChild(btn);
  });

  /* ── Render a floor ──────────────────────────────────── */
  function showFloor(floor) {
    activeFloor = floor.id;
    mapImg.src  = floor.map;

    mapOverlay.querySelectorAll('.hw-marker').forEach(function (m) { m.remove(); });

    floor.locations.forEach(function (loc) {
      var btn = document.createElement('button');
      btn.className = 'hw-marker';
      btn.setAttribute('aria-label', loc.label);
      btn.setAttribute('type', 'button');
      btn.dataset.lx = loc.x;
      btn.dataset.ly = loc.y;

      var lbl = document.createElement('span');
      lbl.className = 'hw-marker-label';
      lbl.textContent = loc.label;
      btn.appendChild(lbl);

      btn.addEventListener('click', function () {
        if (dragMoved) return;
        mapOverlay.querySelectorAll('.hw-marker').forEach(function (m) {
          m.classList.remove('is-active');
        });
        btn.classList.add('is-active');
        floorState[floor.id] = loc.id;
        currentLoc = loc;
        renderPanel(loc);
      });

      mapOverlay.appendChild(btn);
    });

    updateMarkerPositions();
  }

  /* ── Clear panel (no location selected) ─────────────── */
  function clearPanel() {
    panel.classList.add('is-fading');
    setTimeout(function () {
      panelName.textContent = '';
      panelImgWrap.style.display = 'none';
      panelDesc.textContent = 'Click locations to explore';
      panelDesc.style.display = '';
      panelDetails.classList.remove('is-open');
      panelDetails.style.display = 'none';
      panel.classList.remove('is-fading');
    }, 150);
  }

  /* ── Render info panel ───────────────────────────────── */
  function renderPanel(loc) {
    var wasOpen = panelDetails.classList.contains('is-open');
    panel.classList.add('is-fading');

    /* Gate: apply content only after BOTH the panel has faded out
       AND the new image (if any) has finished loading.
       This prevents the old image lingering while the new one loads. */
    var fadeDone = false, imgDone = false, applied = false;

    function maybeApply() {
      if (applied || !fadeDone || !imgDone) return;
      applied = true;

      panelName.textContent = loc.label;

      if (loc.image) {
        panelImg.src = loc.image; /* already in cache — instant render */
        panelImg.alt = loc.label;
        panelImgWrap.style.display = '';
      } else {
        panelImgWrap.style.display = 'none';
      }

      if (loc.desc) {
        panelDesc.textContent = loc.desc;
        panelDesc.style.display = '';
      } else {
        panelDesc.style.display = 'none';
      }

      var firstSeen = loc.firstSeen || '';
      var updatedIn = loc.updatedIn || [];
      var hasDetails = loc.history || firstSeen || updatedIn.length > 0;

      if (hasDetails) {
        panelDetails.style.display = '';
        panelHistory.innerHTML = '';

        if (firstSeen) {
          panelHistory.appendChild(makeTagRow('First seen in', [firstSeen]));
        }
        if (updatedIn.length > 0) {
          panelHistory.appendChild(makeTagRow('Updated in', updatedIn));
        }
        if (loc.history) {
          var p = document.createElement('p');
          p.className = 'hw-panel-history-text';
          p.textContent = loc.history;
          panelHistory.appendChild(p);
        }
        if (wasOpen) panelDetails.classList.add('is-open');
        else panelDetails.classList.remove('is-open');
      } else {
        panelDetails.classList.remove('is-open');
        panelDetails.style.display = 'none';
      }

      panel.classList.remove('is-fading');
    }

    /* Condition 1: panel opacity transition has finished */
    setTimeout(function () { fadeDone = true; maybeApply(); }, 150);

    /* Condition 2: new image is preloaded (or no image needed) */
    if (loc.image) {
      var preload = new Image();
      preload.onload = preload.onerror = function () { imgDone = true; maybeApply(); };
      preload.src = loc.image;
      if (preload.complete) imgDone = true; /* already in browser cache */
    } else {
      imgDone = true;
    }
  }

  function makeTagRow(label, films) {
    var row = document.createElement('div');
    row.className = 'hw-tags-row';
    var lbl = document.createElement('span');
    lbl.className = 'hw-tags-label';
    lbl.textContent = label;
    row.appendChild(lbl);
    films.forEach(function (f) {
      var tag = document.createElement('span');
      tag.className = 'hw-tag hw-tag-' + f;
      tag.textContent = f;
      row.appendChild(tag);
    });
    return row;
  }

  /* ── Init ────────────────────────────────────────────── */
  var grounds = HW_FLOORS.filter(function (f) { return f.id === 'grounds'; })[0];
  if (!grounds) return;

  tabsEl.querySelector('.hw-map-tab').classList.add('is-active');

  var lake = grounds.locations.filter(function (l) { return l.id === 'great-lake'; })[0];
  if (lake) {
    floorState['grounds'] = 'great-lake';
    currentLoc = lake;
  }

  /* ── History toggle — smooth expand/collapse ─────────── */
  panelDetails.querySelector('summary').addEventListener('click', function (e) {
    e.preventDefault();
    panelDetails.classList.toggle('is-open');
  });

  var lastFootnoteTrigger = null;

  document.querySelectorAll('[data-footnote-target]').forEach(function (trigger) {
    trigger.addEventListener('click', function () {
      lastFootnoteTrigger = trigger;
      var targetId = trigger.getAttribute('data-footnote-target');
      var target = document.getElementById(targetId);
      if (!target) return;
      target.scrollIntoView({ behavior: 'smooth', block: 'start' });
    });
  });

  document.querySelectorAll('[data-footnote-back]').forEach(function (button) {
    button.addEventListener('click', function () {
      var targetId = button.getAttribute('data-footnote-back');
      var target = targetId ? document.getElementById(targetId) : null;
      var destination = target || lastFootnoteTrigger;
      if (!destination) return;
      destination.scrollIntoView({ behavior: 'smooth', block: 'center' });
      destination.focus({ preventScroll: true });
    });
  });

  showFloor(grounds);
  if (currentLoc) renderPanel(currentLoc);
  applyTransform();
})();
</script>
